local ui = setmetatable({}, {__index = nui})
local UIX = {VC = {}, HK = {}, dirty = false, focus = nil, listen = nil}

do
  local VC, HK = UIX.VC, UIX.HK
  local VKN = {SPACE = 0x20, TAB = 0x09, SHIFT = 0x10, CTRL = 0x11, ALT = 0x12, CAPS = 0x14,
    LEFT = 0x25, UP = 0x26, RIGHT = 0x27, DOWN = 0x28, INS = 0x2D, DEL = 0x2E, HOME = 0x24,
    END = 0x23, PGUP = 0x21, PGDN = 0x22, M3 = 0x04, M4 = 0x05, M5 = 0x06,
    MOUSE4 = 0x05, MOUSE5 = 0x06}
  for i = 0, 25 do VKN[string.char(65 + i)] = 65 + i end
  for i = 0, 9 do VKN[tostring(i)] = 48 + i; VKN['N' .. i] = 0x60 + i end
  for i = 1, 12 do VKN['F' .. i] = 0x6F + i end

  local VKR = {}
  for name, vk in pairs(VKN) do
    if not VKR[vk] or #name < #VKR[vk] then VKR[vk] = name end
  end
  VKR[0xC0] = '~'; VKR[0xBA] = ';'; VKR[0xBB] = '='; VKR[0xBC] = ','; VKR[0xBD] = '-'
  VKR[0xBE] = '.'; VKR[0xBF] = '/'; VKR[0xDB] = '['; VKR[0xDC] = '\\'; VKR[0xDD] = ']'; VKR[0xDE] = "'"

  function UIX.keyToVk(k)
    if type(k) == 'number' then return k end
    if type(k) ~= 'string' or k == '' or k:lower() == 'none' then return 0 end
    return VKN[k:upper()] or 0
  end
  function UIX.vkName(vk)
    if not vk or vk == 0 then return 'None' end
    return VKR[vk] or string.format('0x%02X', vk)
  end

  function UIX.isKeyDown(vk)
    if not vk or vk == 0 then return false end
    local ok, v = pcall(client.key_state, vk)
    if ok and v == true then return true end
    local name = VKR[vk]
    if name then
      local ok2, v2 = pcall(client.key_state, name)
      if ok2 and v2 == true then return true end
      if name == 'M4' then
        local ok3, v3 = pcall(client.key_state, 'Mouse4')
        if ok3 and v3 == true then return true end
      elseif name == 'M5' then
        local ok3, v3 = pcall(client.key_state, 'Mouse5')
        if ok3 and v3 == true then return true end
      end
    end
    return false
  end

  local function ctl(kind, label, t)
    t.kind, t.label, t.vis, t.group, t.__vc = kind, label, true, 'Misc', true
    VC[#VC + 1] = t
    return t
  end

  function ui.new_checkbox(label) return ctl('check', label, {val = false}) end
  function ui.new_slider(label, mn, mx, def, unit)
    return ctl('slider', label, {min = mn, max = mx, val = clamp(def or mn, mn, mx), unit = unit or ''})
  end
  function ui.new_combobox(label, opts) return ctl('combo', label, {opts = opts, val = 1}) end
  function ui.new_multiselect(label, opts)
    local v = {}
    for i = 1, #opts do v[i] = false end
    return ctl('multi', label, {opts = opts, val = v})
  end
  function ui.new_textbox(label, def) return ctl('text', label, {val = def or ''}) end
  function ui.new_color_picker(label, r, g, b, a)
    return ctl('color', label, {val = {r or 0, g or 175, b or 255, a or 255}})
  end
  function ui.new_hotkey(label, key, mode)
    if mode ~= 'Toggle' and mode ~= 'Always' then mode = 'Hold' end
    local c = ctl('hotkey', label, {vk = UIX.keyToVk(key), mode = mode, active = (mode == 'Always'), prev = false})
    HK[#HK + 1] = c
    return c
  end

  function ui.get(h)
    if type(h) == 'table' and h.__vc then
      local k = h.kind
      if k == 'color' then return h.val[1], h.val[2], h.val[3], h.val[4] end
      if k == 'hotkey' then return h.active end
      return h.val
    end
    return nui.get(h)
  end

  function ui.set(h, a, b, c2, d)
    if type(h) == 'table' and h.__vc then
      local k = h.kind
      if k == 'check' then h.val = a and true or false
      elseif k == 'slider' then h.val = clamp(tonumber(a) or h.val, h.min, h.max)
      elseif k == 'combo' then h.val = clamp(math.floor(tonumber(a) or 1), 1, #h.opts)
      elseif k == 'multi' then
        local v = {}
        for i = 1, #h.opts do v[i] = (type(a) == 'table' and a[i]) and true or false end
        h.val = v
      elseif k == 'text' then h.val = tostring(a or '')
      elseif k == 'color' then h.val = {a or 0, b or 175, c2 or 255, d or 255}
      elseif k == 'hotkey' then
        if type(a) == 'string' or type(a) == 'number' then
          h.vk = UIX.keyToVk(a)
          if b == 'Hold' or b == 'Toggle' or b == 'Always' then h.mode = b end
        elseif type(a) == 'boolean' then
          h.active = a
        end
      end
      UIX.dirty = true
      return
    end
    return nui.set(h, a, b, c2, d)
  end

  function ui.set_visible(h, v)
    if type(h) == 'table' and h.__vc then h.vis = v and true or false; return end
    return nui.set_visible(h, v)
  end
  function ui.set_group(h, g)
    if type(h) == 'table' and h.__vc then h.group = g; return end
  end
  function ui.set_callback(h, fn)
    if type(h) == 'table' and h.__vc then h.cb = fn; return end
    return nui.set_callback(h, fn)
  end

  function UIX.poll()
    if UIX.focus or UIX.listen then return end
    for _, c in ipairs(HK) do
      local down = UIX.isKeyDown(c.vk)
      if c.mode == 'Always' then c.active = true
      elseif c.mode == 'Toggle' then
        if down and not c.prev then c.active = not c.active end
      else c.active = down end
      c.prev = down
    end
  end
end

end

local SAFE_MODS = {2, 3, 6, 7, 8, 9}
local function rerollRound()
  if not ui.get(RR.on) then return end
  local rng = ui.get(RR.yaw)
  local isStaticPreset = (ui.get(PA) == 2)
  for i = 1, 5 do
    local c = aa[i]
    if isStaticPreset then
      c.yaw = nyaw(180 + (math.random() * 2 - 1) * math.min(rng, 10))
      if ui.get(RR.side) then c.side = math.random(1, 2); c.amount = 58; c.dmode = 1 end
    else
      c.yaw = nyaw(180 + (math.random() * 2 - 1) * rng)
      if ui.get(RR.side) then c.side = math.random(1, 2); c.amount = math.random(48, 58) end
      if ui.get(RR.mod) then c.mod = SAFE_MODS[math.random(1, #SAFE_MODS)]; c.range = math.random(28, 64) end
    end
  end
  loadState(curState()); refreshAAVis(); UIX.dirty = true
end

client.set_event_callback('net_update_end', function()
  local aliveSlots = {}
  for _, slot in ipairs(entity.get_players(true)) do
    local e = entity.get(slot)
    if e and e.health and e.health > 0 and e.yaw then
      aliveSlots[slot] = true
      if curOrigin[slot] then prevOrigin[slot] = curOrigin[slot] end
      curOrigin[slot] = e.origin; curHealth[slot] = e.health
      if R.enabled then
        local a = A[slot]
        if not a then
          a = {side = 1, edgeSide = nil, isStatic = true, isJitter = false, jitterRange = 0, air = false, choked = 0}
          A[slot] = a
        end
        a.air = bit.band(e.flags or 1, 1) == 0
        a.choked = e.choked or 0
        local d = adiff(e.foot or e.yaw, e.yaw)
        if math.abs(d) >= 1.5 then a.side = sgn(d) end

        local okH, hist = pcall(entity.get_history, slot, 8)
        if okH and type(hist) == 'table' and #hist >= 4 then
          local altFlips, prevStep, sumAbsStep, steps = 0, 0, 0, 0
          for i = 2, #hist do
            local s1, s0 = hist[i], hist[i - 1]
            if s1 and s0 and s1.yaw and s0.yaw then
              local step = adiff(s1.yaw, s0.yaw)
              if math.abs(step) > 16 and math.abs(prevStep) > 16 and (step * prevStep < 0) then
                altFlips = altFlips + 1
                sumAbsStep = sumAbsStep + math.abs(step)
                steps = steps + 1
              end
              prevStep = step
            end
          end
          a.isJitter = (altFlips >= 2)
          a.isStatic = not a.isJitter
          a.jitterRange = (steps > 0) and (sumAbsStep / steps) or 0
        else
          a.isStatic = true; a.isJitter = false
        end
      end
    else
      curOrigin[slot], curHealth[slot] = nil, nil
    end
  end
  for slot in pairs(curHealth) do
    if not aliveSlots[slot] then curHealth[slot], curOrigin[slot] = nil, nil end
  end
end)

local function angleFor(h, t, md, p, delta, info)
  if h == 1 then
    local s = (info and info.edgeSide) or p
    return nyaw(t.yaw + s * md)
  elseif h == 2 then return nyaw(t.yaw + p * md)
  elseif h == 3 then return nyaw(t.yaw - p * md)
  elseif h == 4 then return nyaw(t.yaw + p * md * 0.55)
  elseif h == 5 then return nyaw(t.yaw - p * md * 0.55)
  elseif h == 6 then return t.yaw
  elseif h == 7 then return t.foot_yaw end
  local q = ((t.tick or 0) % 2 == 0) and 1 or -1
  return nyaw(t.yaw + p * q * md)
end

client.set_event_callback('resolve', function(t)
  if not R.enabled then return nil end
  local st = getState(t.slot)
  local info = A[t.slot]
  local md = math.min((t.max_desync or 58) * R.scale, 58)
  local delta = adiff(t.foot_yaw, t.yaw)

  local p
  if R.mode == 4 then p = R.side
  elseif st.lock and st.lockSide and R.lock then p = st.lockSide
  elseif info and info.edgeSide then p = info.edgeSide
  elseif info and info.side then p = info.side
  else p = (math.abs(delta) >= 1) and sgn(delta) or 1 end

  local h
  if info and info.air and R.air ~= 1 then
    h = (R.air == 2) and 6 or 7
  elseif st.lock and R.lock and R.mode ~= 4 then
    h = st.lock
  elseif R.mode == 1 and info then
    if info.isStatic then
      local staticOrder = {1, 3, 4, 5, 6, 7}
      h = staticOrder[(st.stage - 1) % #staticOrder + 1]
    elseif info.isJitter then
      local jitterOrder = {6, 8, 2, 3, 4, 5}
      h = jitterOrder[(st.stage - 1) % #jitterOrder + 1]
    end
  end

  if not h then
    local order = R.orders[R.mode]
    local n = order and #order or 0
    if n == 0 then return nil end
    h = order[(st.stage - 1) % n + 1]
  end

  st.last = h; st.lastSide = p
  local out = angleFor(h, t, md, p, delta, info)
  st.lastAngle = adiff(out, t.yaw)
  return out
end)

local function isResolverMiss(e)
  local rs = (type(e.reason) == 'string') and e.reason:lower() or nil
  if not rs then return true end
  if rs:find('resolver', 1, true) or rs:find('unknown', 1, true) or rs:find('correction', 1, true) then return true end
  if rs:find('spread', 1, true) or rs:find('occlusion', 1, true)
    or rs:find('death', 1, true) or rs:find('predict', 1, true) then return false end
  return true
end

local function pickKillsay(name)
  local raw = ui.get(KS.text) or ''
  local list = {}
  for part in string.gmatch(raw, '[^|]+') do
    local s = part:match('^%s*(.-)%s*$')
    if s and #s > 0 then list[#list + 1] = s end
  end
  if #list == 0 then return nil end
  local s = list[math.random(1, #list)]
  return (s:gsub('{name}', name or 'enemy'))
end

local SETTING_PREFIXES = {'Rage.', 'AA.', 'Visuals.', 'World.', 'Misc.', ''}
local function listKeys(prefix)
  local out = {}
  local ok, list = pcall(nui.list_settings, prefix)
  if ok and type(list) == 'table' then
    for k, v in pairs(list) do
      local key = nil
      if type(v) == 'string' then key = v elseif type(k) == 'string' then key = k end
      if key then out[#out + 1] = key end
    end
  end
  return out
end

local function findAllKeys(patterns)
  local hits, seen = {}, {}
  for _, prefix in ipairs(SETTING_PREFIXES) do
    for _, key in ipairs(listKeys(prefix)) do
      if not seen[key] then
        seen[key] = true
        local l = key:lower()
        for _, pat in ipairs(patterns) do
          if l:find(pat, 1, true) then hits[#hits + 1] = key; break end
        end
      end
    end
  end
  return hits
end

local tpHandles, dtHandles, btHandles, tagHandles = {}, {}, {}, {}

local function discoverNativeHandles()
  tpHandles, dtHandles, btHandles, tagHandles = {}, {}, {}, {}

    if ok and h then tpHandles[#tpHandles + 1] = {h = h, k = k} end
  end
  for _, k in ipairs(findAllKeys({'doubletap', 'double_tap', 'double tap'})) do
    local ok, h = pcall(nui.reference, k)
    if ok and h then dtHandles[#dtHandles + 1] = {h = h, k = k} end

    local ok, h = pcall(nui.reference, k)
    if ok and h then btHandles[#btHandles + 1] = {h = h, k = k} end
  end
  for _, k in ipairs(findAllKeys({'clantag', 'clan_tag', 'clan tag', 'tag'})) do
    local ok, h = pcall(nui.reference, k)
    if ok and h then tagHandles[#tagHandles + 1] = {h = h, k = k} end
  end
end
discoverNativeHandles()

local function isNativeDtActive()
  if #dtHandles == 0 then return true end
  local anyBoolTrue, hasHotkey, hotkeyActive = false, false, false
  for _, item in ipairs(dtHandles) do
    local ok, cur = pcall(nui.get, item.h)
    if ok then
      if type(cur) == 'boolean' then
        if cur == true then anyBoolTrue = true end
      elseif type(cur) == 'table' and cur.Mode then
        hasHotkey = true
        local m = cur.Mode
        local vk = UIX.keyToVk(cur.Key or '')
        local down = UIX.isKeyDown(vk)
        local tr = dtToggleTracker[item.k] or {state = false, prev = false}
        if down and not tr.prev then tr.state = not tr.state end
        tr.prev = down; dtToggleTracker[item.k] = tr
        if m == 'Always' then hotkeyActive = true
        elseif m == 'Hold' and down then hotkeyActive = true
        elseif m == 'Toggle' and tr.state then hotkeyActive = true end
      end
    end
  end
  if hasHotkey then return hotkeyActive and (anyBoolTrue or #dtHandles == 1) end
  return anyBoolTrue
end

local function setNativeControl(list, wantBool, restoreSaved)
  for _, item in ipairs(list) do
    local h, k = item.h, item.k
    local ok, cur = pcall(nui.get, h)
    if ok then
      if restoreSaved and dtSavedState[k] ~= nil then
        pcall(nui.set, h, dtSavedState[k])
        dtSavedState[k] = nil
      else
        if dtSavedState[k] == nil then
          if type(cur) == 'table' then dtSavedState[k] = {Key = cur.Key, Mode = cur.Mode}
          else dtSavedState[k] = cur end
        end
        if type(cur) == 'boolean' then pcall(nui.set, h, wantBool)
        elseif type(cur) == 'table' and cur.Mode then
          pcall(nui.set, h, {Key = cur.Key or 'None', Mode = wantBool and 'Always' or 'OffHotkey'})
        end
      end
    end
  end
end

local function enforceClanTag()
  if #tagHandles == 0 then discoverNativeHandles() end
  for _, item in ipairs(tagHandles) do
    local h = item.h
    local ok, cur = pcall(nui.get, h)
    if ok then
      if type(cur) == 'string' then
        if cur ~= FORCED_TAG then pcall(nui.set, h, FORCED_TAG) end
      elseif type(cur) == 'boolean' then
        if cur ~= true then pcall(nui.set, h, true) end
      elseif type(cur) == 'number' then
        if cur < 2 then pcall(nui.set, h, 2) end
      end
    end
  end
end

-- ============================================================ Auto Teleport in Air
local airAt, tpFired, tpActive, tpLastAt = nil, false, false, -100
local tpChoked, tpPhase = 0, 0

local function tpWant(realState, now)
  if not ui.get(D.tp) or realState ~= 4 then
    if tpPhase ~= 0 then
      setNativeControl(dtHandles, true, true)
      setNativeControl(tpHandles, false, true)
      tpPhase = 0
    end
    airAt, tpFired, tpActive = nil, false, false
    return false
  end
  if not airAt then airAt = now; tpFired = false end
  local trig = ui.get(D.tpTrig)
  local seenEnough = (now - airAt) * 1000 >= ui.get(D.tpd)
  local cdReady = (now - tpLastAt) * 1000 >= ui.get(D.tpcd)
  local wantByTimer = (trig == 1 or trig == 3) and seenEnough and not tpFired and cdReady
  local wantByVision = (trig == 2 or trig == 3) and seenEnough and exposed and cdReady
  return wantByTimer or wantByVision
end

local function tpTick(realState)
  local now = globals.realtime()
  local want = tpWant(realState, now)
  if not ui.get(D.tp) then return end
  if PK.held or PK.returning then return end

  if want and tpPhase == 0 then
    tpPhase = 1; tpFired = true; tpLastAt = now; tpActive = true; tpChoked = 0
    setNativeControl(dtHandles, false, false)
    setNativeControl(tpHandles, true, false)
  elseif tpPhase == 1 and (now - tpLastAt) >= 0.07 then
    tpPhase = 2
    setNativeControl(dtHandles, true, true)
    setNativeControl(tpHandles, false, true)
  elseif tpPhase == 2 and (now - tpLastAt) >= 0.22 then
    tpPhase = 0; tpActive = false
  end
end

-- ============================================================ Instant 1-Frame Screen-Space Movement Basis (Smart Peek)
local moveBasis = {fx = 0, fz = 1, rx = 1, rz = 0, valid = false}

local function updateMoveBasis()
  local pos = localPos()
  if not pos then return end
  local cx, cy = renderer.world_to_screen(pos.X, pos.Y, pos.Z)
  local x1, y1 = renderer.world_to_screen(pos.X + 24, pos.Y, pos.Z)
  local x2, y2 = renderer.world_to_screen(pos.X, pos.Y, pos.Z + 24)
  if cx and x1 and x2 then
    local f_dx, f_dz = -(y1 - cy), -(y2 - cy)
    local flen = math.sqrt(f_dx * f_dx + f_dz * f_dz)
    local r_dx, r_dz = (x1 - cx), (x2 - cx)
    local rlen = math.sqrt(r_dx * r_dx + r_dz * r_dz)
    if flen > 0.1 and rlen > 0.1 then
      moveBasis.fx, moveBasis.fz = f_dx / flen, f_dz / flen
      moveBasis.rx, moveBasis.rz = r_dx / rlen, r_dz / rlen
      moveBasis.valid = true
      return
    end
  end
  local _, yaw = client.camera_angles()
  if camera and camera.get then
    local ok, cam = pcall(camera.get)
    if ok and cam and cam.yaw then yaw = cam.yaw end
  end
  local rad = math.rad(yaw or 0)
  moveBasis.fx, moveBasis.fz = -math.sin(rad), -math.cos(rad)
  moveBasis.rx, moveBasis.rz = math.cos(rad), -math.sin(rad)
  moveBasis.valid = true
end

local function worldToYaw(dx, dz)
  if camera and camera.get then
    local ok, cam = pcall(camera.get)
    if ok and cam and cam.position and cam.yaw then
      local p = cam.position
      local sw = client.screen_size()
      local bestYaw, bestDist = cam.yaw, 1e9
      for off = 0, 270, 90 do
        local a = math.rad(cam.yaw + off)
        local px = renderer.world_to_screen(p.X + math.cos(a) * 200, p.Y, p.Z + math.sin(a) * 200)
        if px and math.abs(px - sw * 0.5) < bestDist then
          bestDist = math.abs(px - sw * 0.5)
          bestYaw = math.deg(atan2(dz, dx)) - off
        end
      end
      return nyaw(bestYaw)
    end
  end
  return nyaw(math.deg(atan2(dz, dx)))
end

client.set_event_callback('setup_command', function(cmd)
  UIX.poll()
  local held = ui.get(X.peek)
  local now = globals.realtime()
  local mode = ui.get(X.pkret)
  local pos = localPos()

  if held and not PK.held then
    PK.held, PK.returning = true, false
    if pos then PK.anchor = pos end
  elseif held and PK.held and not PK.anchor and pos then
    PK.anchor = pos
  elseif (not held) and PK.held then
    PK.held = false
    if PK.anchor and pos and (mode == 2 or mode == 3) then
      PK.returning, PK.retAt = true, now
    else
      PK.anchor = nil; PK.returning = false
    end
  end

  if not (PK.returning and PK.anchor and pos) then return end

  local dx, dz = PK.anchor.X - pos.X, PK.anchor.Z - pos.Z
  local dist = math.sqrt(dx * dx + dz * dz)
  if dist < 4.5 or (now - PK.retAt > 2.0) then
    PK.returning = false
    cmd.forwardmove = 0; cmd.sidemove = 0
    if not PK.held then PK.anchor = nil end
    return
  end

  local spd = (ui.get(X.pkspd) or 100) / 100
  local scale = math.min(1, dist / 14) * 127 * spd
  local fwd = (dx * moveBasis.fx + dz * moveBasis.fz) / dist
  local str = (dx * moveBasis.rx + dz * moveBasis.rz) / dist
  cmd.forwardmove = clamp(fwd * scale, -127, 127)
  cmd.sidemove    = clamp(str * scale, -127, 127)
end)

client.every(0.08, function()
  local me = entity.get_local_player()
  if not me then return end
  local ox, oy, oz = entity.get_origin(me)
  if not ox then return end
  local myEye = vector.new(ox, oy + 18, oz)
  nearestEnemy, nearestSlot = nil, nil
  local nd = 1e18
  local sw, sh = client.screen_size()
  local cx, cy, cd = sw / 2, sh / 2, 1e18
  crosshairSlot = nil

  for _, slot in ipairs(entity.get_players(true)) do
    local o = curOrigin[slot]
    if o and entity.is_alive(slot) then
      local dd = (o.X - ox) ^ 2 + (o.Z - oz) ^ 2
      if dd < nd then nd, nearestEnemy, nearestSlot = dd, o, slot end
      local sx, sy = renderer.world_to_screen(o.X, o.Y + 20, o.Z)
      if sx then
        local sd = (sx - cx) ^ 2 + (sy - cy) ^ 2
        if sd < cd then cd, crosshairSlot = sd, slot end
      end
    end
  end
  nearestSlot = crosshairSlot or nearestSlot

  local any, seen, n = false, false, 0
  for _, slot in ipairs(entity.get_players(true)) do
    local o = curOrigin[slot]
    if o and entity.is_alive(slot) then
      any = true; n = n + 1
      if n > 4 then break end
      local enemyEye = vector.new(o.X, o.Y + 18, o.Z)
      local tr = client.trace_line(enemyEye, myEye)
      if not tr or not tr.hit or (tr.fraction or 0) >= 0.94 then seen = true end
      local info = A[slot]
      if info and slot == nearestSlot then
        local dx, dz = ox - o.X, oz - o.Z
        local len = math.sqrt(dx * dx + dz * dz)
        if len > 1 then
          local rx, rz = -dz / len, dx / len
          local leftHead  = vector.new(o.X + rx * 18, o.Y + 18, o.Z + rz * 18)
          local rightHead = vector.new(o.X - rx * 18, o.Y + 18, o.Z - rz * 18)
          local trL = client.trace_line(myEye, leftHead)
          local trR = client.trace_line(myEye, rightHead)
          local lBlocked = trL and trL.hit and (trL.fraction or 1) < 0.92
          local rBlocked = trR and trR.hit and (trR.fraction or 1) < 0.92
          if lBlocked and not rBlocked then info.edgeSide = 1
          elseif rBlocked and not lBlocked then info.edgeSide = -1
          else info.edgeSide = nil end
        end
      end
    end
  end
  exposed = (not any) or seen
end)

local manual = nil
local prevKey = {L = false, R = false, B = false}
local function pollManual()
  for _, k in ipairs({'L', 'R', 'B'}) do
    local v = ui.get(K[k])
    if v and not prevKey[k] then
      if manual == k then manual = nil else manual = k end
    end
    prevKey[k] = v
  end
end

local function stateIndex()
  if not isLocalOnGround() then return 4 end
  local s = entity.get_local_state()
  if s and s.move then
    if (s.move.duckAmount or 0) > 0.5 then return 5 end
    if s.move.velocity and s.move.velocity.Magnitude > 5 then return 3 end
  end
  local cond = antiaim and antiaim.get_condition and antiaim.get_condition()
  if cond == 'moving' then return 3 end
  return 2
end

local abFlipV, abYawV = 1, 0
local AB_OPTS = {-18, -12, 12, 18}
abTrigger = function()
  local now = globals.realtime()
  if now - abAt < 0.2 then return end
  abAt = now; abStage = abStage + 1; abFlipV = -abFlipV
  local y, guard = abYawV, 0
  while y == abYawV and guard < 8 do
    y = AB_OPTS[math.random(1, #AB_OPTS)]; guard = guard + 1
  end
  abYawV = y
end

client.set_event_callback('bullet_impact', function(e)
  if not ui.get(E.aa) or not ui.get(K.ab) then return end
  local me = entity.get_local_player()
  if not me or e.shooter == me or not entity.is_enemy(e.shooter) then return end
  if not e.start or not e.impact then return end
  local ox, oy, oz = entity.get_origin(me)
  if not ox then return end
  local p = vector.new(ox, oy + 18, oz)
  local ab = e.impact - e.start
  local len2 = ab:Dot(ab)
  if len2 < 0.001 then return end
  local t = clamp((p - e.start):Dot(ab) / len2, 0, 1)
  local closest = e.start + ab * t
  if (p - closest).Magnitude < 55 then abTrigger() end
end)

local flChoked, flTarget = 0, nil
local DEF_PAT = {{4, 2}, {8, 3}}
local defStartPk, defNextPk = 0, 0
local function isDefActive(pk, choked, pat)
  if pat == 1 then return (choked == 0) or (pk % 3 == 0)
  elseif pat == 2 or pat == 3 then
    local t = DEF_PAT[pat - 1]
    return pk % (t[1] + t[2]) < t[2]
  end
  if pk < defStartPk - 40 then defStartPk, defNextPk = 0, 0 end
  if pk >= defNextPk then
    defStartPk = pk + math.random(2, 8)
    defNextPk = defStartPk + math.random(2, 4)
  end
  return pk >= defStartPk and pk < defNextPk
end

local lastAA = {state = 2, side = 1, amount = 58, manual = nil, def = false, dtActive = false}
local prevPacket, rndYaw, rndSide, rndP = -1, 0, 1, 0

client.set_event_callback('anti_aim', function(cmd)
  UIX.poll()
  if not ui.get(E.aa) then defActive = false; return end
  pollManual()
  local real = stateIndex()
  local idx = real
  local c = aa[idx]
  if not c.override then idx = 1; c = aa[1] end

  local pk = cmd.packet or 0
  local choked = cmd.chokedcommands or 0
  if pk ~= prevPacket then
    prevPacket = pk
    rndYaw = math.random() * 2 - 1
    rndP = math.random() * 2 - 1
    rndSide = (math.random(0, 1) == 0) and -1 or 1
  end
  local phase = (pk % 2 == 0) and 1 or -1
  local now = globals.realtime()
  local range, speed = c.range, c.speed

  local off, dir = 0, phase
  local mod = c.mod
  if mod == 2 then off = phase * range * 0.5
  elseif mod == 3 then off = (phase == 1) and range or -range * 0.25
  elseif mod == 4 then off = rndYaw * range * 0.5; dir = sgn(rndYaw)
  elseif mod == 5 then
    local spin = (now * speed * 8) % 360
    off = spin; dir = (spin < 180) and 1 or -1
  elseif mod == 6 then
    local v = (pk % 3) - 1; off = v * range * 0.5; if v ~= 0 then dir = v end
  elseif mod == 7 then
    local v = (pk % 5) - 2; off = v * range * 0.25; if v ~= 0 then dir = sgn(v) end
  elseif mod == 8 then
    local s = math.sin(now * speed * 0.15); off = s * range * 0.5; dir = sgn(s)
  elseif mod == 9 then
    off = (phase == 1) and (range * 0.45) or (-range * 0.18); dir = phase
  end

  local abOn = ui.get(K.ab)
  local isPureStatic = (mod == 1 and c.dmode == 1)
  local abYaw = (abOn and not isPureStatic) and abYawV or 0
  local abFlip = abOn and abFlipV or 1

  local yawOut
  if manual then
    if manual == 'L' then yawOut = 90
    elseif manual == 'R' then yawOut = -90
    else yawOut = 180 end
  else
    yawOut = c.yaw + off + abYaw
  end

  local baseYaw = cmd.view_yaw
  if (not manual) and ui.get(E.base) == 2 and nearestEnemy then
    local mp = localPos()
    if mp then baseYaw = worldToYaw(nearestEnemy.X - mp.X, nearestEnemy.Z - mp.Z) end
  end
  cmd.yaw = nyaw(baseYaw + yawOut)

  local p = c.pitch
  if p == 2 then cmd.pitch = 89
  elseif p == 3 then cmd.pitch = -89
  elseif p == 4 then cmd.pitch = 0
  elseif p == 5 then cmd.pitch = 89 * phase
  elseif p == 6 then cmd.pitch = math.floor(rndYaw * 89) end

  local dm = c.dmode
  local side
  if dm == 2 then side = phase
  elseif dm == 3 then side = rndSide
  elseif dm == 4 then side = -dir
  elseif dm == 6 then
    side = (c.side == 1) and 1 or -1
    if pk % 3 == 0 then side = -side end
  else side = (c.side == 1) and 1 or -1 end

  local amt = c.amount
  if dm == 5 then amt = amt * (0.6 + 0.4 * (math.sin(now * speed * 0.3) * 0.5 + 0.5))
  elseif dm == 6 then amt = (phase == 1) and 58 or 31 end
  local jit = ui.get(E.djit)
  if jit > 0 then amt = clamp(amt + rndP * jit, 0, 58) end
  if ui.get(K.inv) then side = -side end
  side = side * abFlip
  if side == 0 then side = 1 end

  defActive = false
  local dtNow = isNativeDtActive()
  lastAA.dtActive = dtNow
  local dtAllowed = (not ui.get(D.dtOnly)) or dtNow
  local defWhere = ui.get(D.where)
  local stateAllowed = (defWhere == 3) or (defWhere == 2 and (real == 4 or real == 3)) or (defWhere == 1 and real == 4)
  if ui.get(D.on) and dtAllowed and stateAllowed and not manual and exposed then
    if isDefActive(pk, choked, ui.get(D.pat)) then
      defActive = true
      local sens = ui.get(D.sens)
      cmd.yaw = nyaw(baseYaw + ui.get(D.yaw) * phase + rndYaw * sens * 1.4)
      cmd.pitch = clamp(ui.get(D.pitch) + rndP * sens * 0.25, -89, 89)
      side = -side; amt = 58
    end
  end

  cmd.desync_amount = clamp(amt, 0, 58)
  cmd.desync_side = side

  lastAA.state, lastAA.side, lastAA.amount = idx, side, cmd.desync_amount
  lastAA.manual, lastAA.def = manual, defActive

  if tpPhase == 1 and real == 4 and ui.get(D.tp) then
    local n = ui.get(D.tpChoke)
    if tpChoked < n then cmd.choke = true; tpChoked = tpChoked + 1
    else cmd.send_packet = true; tpChoked = 0; tpPhase = 2 end
  elseif ui.get(FL.on) then
    local maxN, var = ui.get(FL.max), ui.get(FL.var)
    if not flTarget then flTarget = clamp(maxN + math.random(-var, var), 1, 14) end
    if flChoked < flTarget then cmd.choke = true; flChoked = flChoked + 1
    else cmd.send_packet = true; flChoked, flTarget = 0, nil end
  end
end)

-- ============================================================ Pool-Safe 2D Drawing Primitives
local function RRect(x, y, w, h, r, g, b, a, rad)
  renderer.rounded_rectangle(x, y, w, h, r, g, b, a, math.max(1, rad or 1))
end

local function RText(x, y, s, size, font, col, a, align, outline)
  renderer.text_ex(x, y, tostring(s or ''), {
    font = font or 'ui_bold',
    size = size or 13,
    align = align or 'left',
    outline = (outline == true),
    rotation = 0,
    color = {col[1] or 220, col[2] or 230, col[3] or 245, a or col[4] or 255}
  })
end

drawBinds = function(sh, r, g, b)

  local list = {
    {'Desync inverter', ui.get(K.inv) and 'on' or nil},
    {'Manual left', (m == 'L') and 'on' or nil},
    {'Manual right', (m == 'R') and 'on' or nil},

  }

  local k = 1 - math.exp(-dt * 14)

    local name, st = it[1], it[2]
    if st then bindState[name] = st end

      local rw = renderer.measure_text(name, 12, 'ui_bold') + renderer.measure_text(bindState[name] or '', 12, 'ui_bold') + 42
      wantW = math.max(wantW, rw)
    end
  end
  if maxA < 0.02 then return end
  bindW = bindW + (wantW - bindW) * k
  local x, y = 20, math.floor(sh * 0.27)
  local w, h = bindW, 28 + rowsH + 4
  local A = maxA

local HIT, MISS, KILL = {125, 230, 160}, {255, 110, 120}, {255, 205, 100}
drawHitlog = function(sh)
  local now = globals.realtime()
  for i = #rows, 1, -1 do
    if now - rows[i].at >= LOG_LIFE then table.remove(rows, i) end
  end
  local y0 = math.floor(sh * 0.54)
  for i, row in ipairs(rows) do
    local age = now - row.at
    local fadeOut = clamp((LOG_LIFE - age) / 0.4, 0, 1)
    local slide = clamp(age / 0.18, 0, 1)
    local alpha = math.floor(255 * fadeOut * slide)
    local col = (row.kind == 'hit' and HIT) or (row.kind == 'kill' and KILL) or MISS
    local w = renderer.measure_text(row.text, 13, 'ui_bold') + 28
    local x = 20 - (1 - slide) * 24
    local y = y0 + (i - 1) * 28
    RRect(x, y, w, 24, 8, 14, 22, math.floor(alpha * 0.9), 6)
    RRect(x, y, 4, 24, col[1], col[2], col[3], alpha, 2)
    RText(x + 14, y + 5, row.text, 13, 'ui_bold', col, alpha, 'left', false)
  end
end

local function drawTags(cx, y, tags, r, g, b)

  for i, t in ipairs(tags) do
    widths[i] = renderer.measure_text(t.text, 11, 'ui_bold') + 14
    total = total + widths[i] + 4
  end
  local x = cx - (total - 4) / 2
  for i, t in ipairs(tags) do

    RText(x + 7, y + 2, t.text, 11, 'ui_bold', c, 255, 'left', false)
    x = x + widths[i] + 4
  end
end

drawKillPopup = function(sw, sh, r, g, b)
  if not kstreak.popup then return end
  local age = globals.realtime() - kstreak.popupAt
  local life = 1.8
  if age > life then kstreak.popup = nil; return end
  local grow = clamp(age / 0.18, 0, 1)
  local size = math.floor(16 + 18 * grow)
  local fade = (age > life - 0.5) and clamp((life - age) / 0.5, 0, 1) or 1
  local a = math.floor(255 * fade)
  local y = sh * 0.14
  RText(sw / 2, y, kstreak.popup, size, 'gotham', kstreak.popupColor, a, 'center', true)

local function trimText(s, maxw, size, font)
  if renderer.measure_text(s, size, font) <= maxw then return s end
  local guard = 0
  while #s > 1 and guard < 40 and renderer.measure_text(s .. '..', size, font) > maxw do
    s = s:sub(1, #s - 1); guard = guard + 1
  end
  return s .. '..'
end

local function panelCorner(sw, sh, w, h, corner)
  local pad = 22
  local x = (corner == 1 or corner == 3) and pad or (sw - w - pad)
  local y = (corner == 1 or corner == 2) and sh * 0.22 or (sh - h - pad)
  return x, y
end

local function confColor(st)
  local hh = st and (st.lock or st.last)
  local conf = hh and hypRate(hh) or 0
  if st and st.lock then return 120, 230, 160, 'LOCKED', conf
  elseif not hh or (st.hits + st.misses) < 1 then return 130, 175, 255, 'READY S' .. (st and st.stage or 1), conf
  elseif conf >= 0.6 then return 170, 225, 115, 'HIGH CONF', conf
  elseif conf >= 0.4 then return 255, 205, 100, 'BRUTE S' .. st.stage, conf
  else return 255, 110, 120, 'BRUTE S' .. st.stage, conf end
end

local CFG = {times = {}}
do
  local function copy(v)
    if type(v) ~= 'table' then return v end
    local t = {}
    for k, x in pairs(v) do t[k] = copy(x) end
    return t
  end
  local function same(a, b)
    if type(a) ~= type(b) then return false end
    if type(a) ~= 'table' then return a == b end
    for k, v in pairs(a) do if not same(v, b[k]) then return false end end
    for k in pairs(b) do if a[k] == nil then return false end end
    return true
  end
  local function ckey(c) return c.group .. '/' .. c.label end
  local function cval(c)
    if c.kind == 'hotkey' then return {c.vk, c.mode} end
    return c.val
  end

  local AA_DEF = copy(aa)
  for _, c in ipairs(UIX.VC) do c.def = copy(cval(c)) end

  function CFG.serialize(full)
    local v = {}
    for _, c in ipairs(UIX.VC) do
      if not c.nocfg then
        local val = cval(c)
        if full or not same(val, c.def) then v[ckey(c)] = copy(val) end
      end
    end
    local a = {}
    for i = 1, 5 do
      if full or not same(aa[i], AA_DEF[i]) then a[tostring(i)] = copy(aa[i]) end
    end
    return utils.json_encode({v = v, aa = a, t = os.date('%H:%M')})
  end

  function CFG.applyTable(t)
    if type(t) ~= 'table' then return false end
    if type(t.v) == 'table' then
      for _, c in ipairs(UIX.VC) do
        local x = (not c.nocfg) and t.v[ckey(c)] or nil
        local k = c.kind
        if x ~= nil then
          if k == 'hotkey' then
            if type(x) == 'table' then
              c.vk = tonumber(x[1]) or c.vk
              if x[2] == 'Hold' or x[2] == 'Toggle' or x[2] == 'Always' then c.mode = x[2] end
              c.active, c.prev = (c.mode == 'Always'), false
            end
          elseif k == 'color' then
            if type(x) == 'table' then ui.set(c, x[1], x[2], x[3], x[4]); c.h = nil end
          elseif k == 'check' then
            if type(x) == 'boolean' then ui.set(c, x) end
          elseif k == 'slider' or k == 'combo' then
            if type(x) == 'number' then ui.set(c, x) end
          elseif k == 'multi' then
            if type(x) == 'table' then ui.set(c, x) end
          elseif k == 'text' then
            if type(x) == 'string' then ui.set(c, x) end
          end
        end
      end
    end
    if type(t.aa) == 'table' then
      for i = 1, 5 do
        local s = t.aa[tostring(i)] or t.aa[i]
        if type(s) == 'table' then
          for kf, v in pairs(aa[i]) do
            if type(s[kf]) == type(v) then aa[i][kf] = s[kf] end
          end
        end
      end
    end
    return true
  end

  function CFG.reset()
    for _, c in ipairs(UIX.VC) do
      if not c.nocfg then
        local d = c.def
        if c.kind == 'hotkey' then
          c.vk, c.mode, c.active, c.prev = d[1], d[2], (d[2] == 'Always'), false
        elseif c.kind == 'color' then
          ui.set(c, d[1], d[2], d[3], d[4]); c.h = nil
        else
          ui.set(c, copy(d))
        end
      end
    end
    for i = 1, 5 do
      for k, v in pairs(AA_DEF[i]) do aa[i][k] = v end
    end
  end

  function CFG.refresh()
    if UIX.afterApply then pcall(UIX.afterApply) end
  end

  function CFG.save(i)
    local ok, s = pcall(CFG.serialize, false)
    if not ok then return false end
    pcall(storage.set, 'cfg_' .. i, s)
    CFG.times[i] = os.date('%H:%M')
    return true
  end

  function CFG.load(i)
    local raw = storage.get('cfg_' .. i)
    if type(raw) ~= 'string' then return false end
    local ok, t = pcall(utils.json_decode, raw)
    if not ok or type(t) ~= 'table' then return false end
    CFG.reset(); CFG.applyTable(t); CFG.refresh()
    UIX.dirty = true
    return true
  end

  function CFG.export()
    local s = CFG.serialize(false)
    client.log('[config] ' .. s)
    client.log('[config] вставь строку выше между [=[ и ]=] в EMBEDDED_CONFIG и нажми Save source')
  end

  function CFG.init()
    local e = EMBEDDED_CONFIG:match('^%s*(.-)%s*$')
    if e and #e > 2 then
      local ok, t = pcall(utils.json_decode, e)
      if ok and type(t) == 'table' then CFG.applyTable(t) end
    end
    local auto = storage.get('cfg_auto')
    if type(auto) == 'string' then
      local ok, t = pcall(utils.json_decode, auto)
      if ok and type(t) == 'table' then CFG.applyTable(t) end
    end
    ui.set(D.tp, false)
    for i = 1, 5 do
      local raw = storage.get('cfg_' .. i)
      if type(raw) == 'string' then
        local ok, t = pcall(utils.json_decode, raw)
        if ok and type(t) == 'table' then CFG.times[i] = t.t or '?' end
      end
    end
    CFG.refresh()
    UIX.dirty = false
  end

  client.every(2, function()
    if UIX.dirty then
      UIX.dirty = false
      local ok, s = pcall(CFG.serialize, true)
      if ok then pcall(storage.set, 'cfg_auto', s) end
    end
  end)
end

local function hsv2rgb(h, s, v)
  local i = math.floor(h * 6) % 6
  local f = h * 6 - math.floor(h * 6)
  local p, q, t2 = v * (1 - s), v * (1 - f * s), v * (1 - (1 - f) * s)
  local r, g, b
  if i == 0 then r, g, b = v, t2, p elseif i == 1 then r, g, b = q, v, p
  elseif i == 2 then r, g, b = p, v, t2 elseif i == 3 then r, g, b = p, q, v
  elseif i == 4 then r, g, b = t2, p, v else r, g, b = v, p, q end
  return math.floor(r * 255 + 0.5), math.floor(g * 255 + 0.5), math.floor(b * 255 + 0.5)
end
local function rgb2hsv(r, g, b)
  r, g, b = r / 255, g / 255, b / 255
  local mx, mn = math.max(r, g, b), math.min(r, g, b)
  local d = mx - mn
  local h = 0
  if d > 0 then
    if mx == r then h = ((g - b) / d) % 6 elseif mx == g then h = (b - r) / d + 2 else h = (r - g) / d + 4 end
    h = h / 6
  end
  return h, (mx == 0) and 0 or d / mx, mx
end

local function keyDown(vk) return UIX.isKeyDown(vk) end

function MN.capture()
  local c = UIX.listen
  if not MN.kpInit then
    MN.kp = {}
    for vk = 3, 254 do if vk ~= 0x02 and keyDown(vk) then MN.kp[vk] = true end end
    MN.kpInit = true
    return
  end
  for vk = 3, 254 do
    if vk ~= 0x02 then
      local d = keyDown(vk)
      if d and not MN.kp[vk] then
        c.vk = (vk == 0x1B) and 0 or vk
        c.prev = true; UIX.listen = nil; changed(c)
        return
      end
      MN.kp[vk] = d
    end
  end
end

function MN.typing()
  local isSearch = MN.searchFocus
  local c = UIX.focus
  if not isSearch and not c then return end
  if not MN.kpInit then
    MN.kp = {}
    for vk in pairs(TYPE) do if keyDown(vk) then MN.kp[vk] = true end end
    MN.kp[0x08], MN.kp[0x0D], MN.kp[0x1B] = keyDown(0x08), keyDown(0x0D), keyDown(0x1B)
    MN.kpInit = true
    return
  end
  local shift = keyDown(0x10)
  local keys = {0x08, 0x0D, 0x1B}
  for vk in pairs(TYPE) do keys[#keys + 1] = vk end
  for _, vk in ipairs(keys) do
    local d = keyDown(vk)
    if d and not MN.kp[vk] then
      if isSearch then
        if vk == 0x08 then MN.search = MN.search:sub(1, #MN.search - 1)
        elseif vk == 0x0D or vk == 0x1B then MN.searchFocus = false
        elseif #MN.search < 32 then MN.search = MN.search .. TYPE[vk][shift and 2 or 1] end
      else
        if vk == 0x08 then c.val = c.val:sub(1, #c.val - 1); changed(c)
        elseif vk == 0x0D or vk == 0x1B then UIX.focus = nil
        elseif #c.val < 160 then
          c.val = c.val .. TYPE[vk][shift and 2 or 1]; changed(c)
        end
      end
    end
    MN.kp[vk] = d
  end

  local hov = inRect(I.mx, I.my, x, y, w, h)
  local clicked = false
  if hov and I.pressed then clicked = true; I.pressed = false end
  RRect(x, y, w, h, hot and 16 or (hov and 14 or 8), hot and 48 or (hov and 26 or 15), hot and 78 or (hov and 40 or 24), 245, 5)
  if iconId then
    drawIcon(iconId, x + 10, y + math.floor((h - 14) / 2), 14, ar, ag, ab, 255)
    T(x + 30, y + math.floor((h - 13) / 2) - 1, text, 13, CT, 'ui_bold')
  else
    T(x + w / 2 - TW(text, 13, 'ui_bold') / 2, y + math.floor((h - 13) / 2) - 1, text, 13, CT, 'ui_bold')
  end

end

function MN.groupPanel(name, x, y, w, ar, ag, ab, filter)
  local list = {}
  for _, c in ipairs(UIX.VC) do
    if c.group == name and c.vis then
      if not filter or filter == '' or c.label:lower():find(filter, 1, true) or name:lower():find(filter, 1, true) then
        list[#list + 1] = c
      end
    end
  end
  if #list == 0 then return y end
  local hgt = 44
  for _, c in ipairs(list) do hgt = hgt + RH[c.kind] end
  RRect(x, y, w, hgt, 8, 14, 22, 232, 7)
  T(x + 14, y + 10, name, 15, {242, 247, 255}, 'ui_bold')
  RRect(x + 12, y + 34, w - 24, 1, 16, 28, 42, 255, 1)
  local ry = y + 38
  for _, c in ipairs(list) do
    ROW[c.kind](c, x, ry, w, ar, ag, ab)
    ry = ry + RH[c.kind]
  end
  return y + hgt + 14
end

function MN.configsPage(x, y, w, ar, ag, ab)
  RRect(x, y, w, 250, 8, 14, 22, 232, 7)
  T(x + 14, y + 10, 'Config slots', 15, {242, 247, 255}, 'ui_bold')
  RRect(x + 12, y + 34, w - 24, 1, 16, 28, 42, 255, 1)
  for i = 1, 5 do
    local ry = y + 42 + (i - 1) * 33
    local active = (MN.slot == i)
    if active then RRect(x + 10, ry - 2, w - 20, 29, 11, 52, 82, 180, 5) end
    if I.pressed and inRect(I.mx, I.my, x + 10, ry - 2, w - 240, 29) then MN.slot = i; I.pressed = false end
    T(x + 18, ry + 5, 'Slot #' .. i, 13, active and {ar, ag, ab} or CT, 'ui_bold')
    local st = CFG.times[i] and ('saved ' .. CFG.times[i]) or 'empty'
    T(x + 96, ry + 6, st, 12, CD, 'ui_bold')
    if MN.button(x + w - 168, ry, 72, 25, 'Save', ar, ag, ab, false) then
      MN.slot = i
      MN.say(CFG.save(i) and ('Saved to slot ' .. i) or 'Save failed')
    end
    if MN.button(x + w - 88, ry, 72, 25, 'Load', ar, ag, ab, false) then
      MN.slot = i
      MN.say(CFG.load(i) and ('Loaded slot ' .. i) or ('Slot ' .. i .. ' is empty'))
    end
  end
  local by = y + 42 + 5 * 33 + 6
  if MN.button(x + 14, by, 156, 27, 'Export to console', ar, ag, ab, false) then
    CFG.export(); MN.say('Exported: see Lua console')
  end
  local now = globals.realtime()
  local armed = now < MN.resetUntil
  if MN.button(x + 180, by, 176, 27, armed and 'Click again to confirm' or 'Reset to defaults', ar, ag, ab, armed) then
    if armed then
      CFG.reset(); CFG.refresh(); UIX.dirty = true
      MN.resetUntil = 0; MN.say('Reset to defaults')
    else
      MN.resetUntil = now + 3
    end
  end
end

function MN.idle()
  MN.dd, MN.active, MN.cdrag, MN.drag = nil, nil, nil, false
  UIX.focus, UIX.listen, MN.searchFocus = nil, nil, false
  MN.prevDown = client.key_state('Mouse1') == true
end

function MN.draw(sw, sh, ar, ag, ab)
  local mx, my = client.mouse_position()
  I.mx, I.my = mx or 0, my or 0
  local down = client.key_state('Mouse1') == true
  local rdown = client.key_state('Mouse2') == true
  I.pressed, I.rpressed, I.down = down and not MN.prevDown, rdown and not MN.prevR, down
  MN.prevDown, MN.prevR = down, rdown
  if not down then MN.active = nil end

  if not MN.x then
    local sx, sy = storage.get('menu_x'), storage.get('menu_y')
    if type(sx) == 'number' and type(sy) == 'number' then
      MN.x, MN.y = sx, sy
    else
      MN.x, MN.y = math.floor((sw - MN.w) * 0.5), math.floor((sh - MN.h) * 0.5)
    end
  end
  MN.x = clamp(MN.x, 0, math.max(0, sw - MN.w))
  MN.y = clamp(MN.y, 0, math.max(0, sh - MN.h))
  local x, y, w, h, sideW, topH = MN.x, MN.y, MN.w, MN.h, MN.sideW, MN.topH

  if MN.drag then
    if down then
      MN.x = clamp(I.mx - MN.dx, 0, math.max(0, sw - w))
      MN.y = clamp(I.my - MN.dy, 0, math.max(0, sh - h))
      x, y = MN.x, MN.y
    else
      MN.drag = false
      pcall(storage.set, 'menu_x', MN.x); pcall(storage.set, 'menu_y', MN.y)
    end
  end

  if UIX.listen then MN.capture() elseif UIX.focus or MN.searchFocus then MN.typing() end
  MN.overlayInput()

  -- 1. Deep Navy-Black Main Frame (#05080D) + Sidebar (#070D14)
  RRect(x, y, w, h, 5, 8, 13, 240, 9)
  RRect(x, y, sideW, h, 7, 13, 20, 248, 9)
  RRect(x + sideW, y, 1, h, 14, 26, 38, 255, 1)
  RRect(x + sideW, y + topH, w - sideW, 1, 14, 26, 38, 255, 1)

  -- 2. Large Glowing ANTINEVERLOSE Logo at Top-Left (21px Gotham)
  T(x + 17, y + 18, 'ANTINEVERLOSE', 21, {ar, ag, ab}, 'gotham', 200)
  T(x + 15, y + 16, 'ANTINEVERLOSE', 21, {ar, ag, ab}, 'gotham', 150)
  T(x + 16, y + 17, 'ANTINEVERLOSE', 21, {245, 250, 255}, 'gotham', 255)

  -- 3. Sidebar Categories & Tab Items with Authentic 16x16 Neverlose Icons
  local ty = y + 66
  for i, tab in ipairs(TABS) do
    if tab.cat then
      ty = ty + 6
      T(x + 18, ty, tab.cat, 12, {86, 102, 122}, 'ui_bold')
      ty = ty + 22
    end
    local hov = inRect(I.mx, I.my, x + 10, ty, sideW - 20, 34)
    if hov and I.pressed then
      MN.tab = i; MN.dd = nil; MN.search = ''; MN.searchFocus = false; I.pressed = false
    end
    local sel = (MN.tab == i)
    if sel then
      RRect(x + 10, ty, sideW - 20, 34, 17, 52, 82, 255, 6)
    elseif hov then
      RRect(x + 10, ty, sideW - 20, 34, 12, 24, 38, 200, 6)
    end
    drawIcon(tab.icon, x + 20, ty + 9, 16, ar, ag, ab, 255)
    T(x + 44, ty + 8, tab.name, 14, sel and {245, 250, 255} or (hov and CT or {185, 198, 215}), 'ui_bold')

    if tab.badge then
      local bw2 = TW(tab.badge, 11, 'ui_bold') + 14
      local bx2 = x + sideW - 18 - bw2
      RRect(bx2, ty + 7, bw2, 20, 14, 44, 70, 255, 5)
      T(bx2 + 7, ty + 9, tab.badge, 11, {ar, ag, ab}, 'ui_bold')
    end
    ty = ty + 38
  end

  -- 4. Sidebar Bottom Profile Footer
  local footY = y + h - 58
  RRect(x, footY, sideW, 1, 14, 26, 38, 255, 1)
  local me = entity.get_local_player()
  local nm = (me and entity.get_name(me)) or 'Player'
  RRect(x + 14, footY + 13, 32, 32, 14, 26, 40, 255, 16)
  T(x + 25, footY + 21, string.upper(nm:sub(1, 1)), 15, {240, 245, 255}, 'ui_bold')
  RRect(x + 38, footY + 36, 8, 8, 76, 217, 100, 255, 4)
  T(x + 54, footY + 13, fit(nm, 162, 13, 'ui_bold'), 13, {240, 245, 255}, 'ui_bold')
  T(x + 54, footY + 31, 'Till: ', 12, {98, 112, 132}, 'ui_bold')
  T(x + 84, footY + 31, 'Lifetime', 12, {ar, ag, ab}, 'ui_bold')

  -- 5. Top Header Bar: Floppy Save Button + Slot Button + Search Box + Gear Icon
  if MN.button(x + sideW + 18, y + 13, 86, 30, 'Save', ar, ag, ab, false, ICONS.save) then
    MN.say(CFG.save(MN.slot) and ('Saved #' .. MN.slot) or 'Failed')
  end
  if MN.button(x + sideW + 112, y + 13, 82, 30, 'Slot #' .. MN.slot, ar, ag, ab, false) then
    MN.slot = MN.slot % 5 + 1
  end

  local sbx, sby, sbw, sbh = x + sideW + 204, y + 13, 210, 30
  if I.pressed and inRect(I.mx, I.my, sbx, sby, sbw, sbh) then
    MN.searchFocus, UIX.focus, UIX.listen, MN.kpInit = true, nil, nil, false
    I.pressed = false
  end
  RRect(sbx, sby, sbw, sbh, MN.searchFocus and 14 or 8, MN.searchFocus and 30 or 15, MN.searchFocus and 46 or 24, 255, 5)
  drawIcon(ICONS.search, sbx + 9, sby + 8, 14, ar, ag, ab, 220)
  local sDisp = (#MN.search > 0 or MN.searchFocus) and MN.search or 'Search...'
  if MN.searchFocus and math.floor(globals.realtime() * 2) % 2 == 0 then sDisp = sDisp .. '|' end
  T(sbx + 29, sby + 7, sDisp, 13, (#MN.search > 0 or MN.searchFocus) and CT or {98, 112, 132}, 'ui_bold')

  local gx0, gy0 = x + w - 34, y + 20
  if I.pressed and inRect(I.mx, I.my, gx0 - 6, gy0 - 6, 28, 28) then
    MN.tab = #TABS; MN.dd = nil; I.pressed = false
  end
  drawIcon(ICONS.gear, gx0, gy0, 16, 200, 212, 228, 255)

  if globals.realtime() - MN.msgAt < 3 then
    T(sbx + sbw + 12, y + 20, MN.msg, 13, {ar, ag, ab}, 'ui_bold')
  end

  if I.pressed and (inRect(I.mx, I.my, x, y, sideW, 60) or inRect(I.mx, I.my, sbx + sbw + 4, y, w - (sbx - x + sbw) - 44, topH)) then
    MN.drag, MN.dx, MN.dy = true, I.mx - x, I.my - y
    I.pressed = false
  end

  -- 6. Content Area
  local cx0, cy0 = x + sideW + 18, y + topH + 16
  local contentW = w - sideW - 36
  local colW = math.floor((contentW - 16) / 2)
  local filter = (#MN.search > 0) and MN.search:lower() or nil

  local tab = TABS[MN.tab]
  if filter then
    local allGroups = {'Resolver', 'Ragebot', 'Points', 'Main AA', 'Modifiers', 'Desync', 'Fake lag',
                       'Binds & reactive', 'Anti-resolver', 'Defensive', 'Smart peek', 'Teleport', 'Visuals', 'Killsay'}
    local gyL, gyR = cy0, cy0
    for idx, g in ipairs(allGroups) do
      if idx % 2 == 1 then
        gyL = MN.groupPanel(g, cx0, gyL, colW, ar, ag, ab, filter)
      else
        gyR = MN.groupPanel(g, cx0 + colW + 16, gyR, colW, ar, ag, ab, filter)
      end
    end
  elseif tab.special then
    MN.configsPage(cx0, cy0, contentW, ar, ag, ab)
  else
    for ci, groups in ipairs(tab.cols) do
      local gx = cx0 + (ci - 1) * (colW + 16)
      local gy = cy0
      for _, g in ipairs(groups) do
        gy = MN.groupPanel(g, gx, gy, colW, ar, ag, ab, nil)
      end
    end
  end

  MN.overlayDraw(ar, ag, ab)

  if I.pressed then
    MN.dd = nil
    UIX.focus, UIX.listen, MN.searchFocus = nil, nil, false
  end
end
end
